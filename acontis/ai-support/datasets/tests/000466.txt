Question About EC-Master(Tokyo Robotics)

Customers refer to "EC-Master_ClassA.pdf - 3.2 Programmer’s Guide" to test FOE.
EcatFoeFileDownload is already working under the condition that they do not use NOTIFY.
They do not have a FOE sample when using NOTIFY.

  "EC-Master-V3.1.1.01-Xenomai3-x86_64Bit-SDK_Protected/Examples/Common/EcNotification.cpp"

  - ecatFoeDownloadReq
  - ecatFoeSegmentedDownloadReq

  - "ecatFoeFileDownload - pbyData"

Could you please provide some sample code to understand these?

<p>/*
ecatFoeFileDownloadReq() */</p>

<p>{</p>

<p>    EC_T_CHAR* szFileName     = "<a href="file://TEST.txt">\\TEST.txt</a>";</p>

<p>    EC_T_CHAR* szDownloadName =
"TEST";</p>

<p>    FILE*     
pFile          = NULL;</p>

<p>    EC_T_DWORD dwDataLen      = 200*1024;</p>

<p>    EC_T_BYTE* pbyFileData    = EC_NULL;</p>

<p>    EC_T_MBXTFER_DESC MbxTferDesc;</p>

<p>    EC_T_MBXTFER*     pMbxTfer   
= EC_NULL;</p>

<p> </p>

<p>    dwRes = ecatSetSlaveState(0,
DEVICE_STATE_BOOTSTRAP, 5000);</p>

<p>    if (dwRes != EC_E_NOERROR)</p>

<p>    {</p>

<p>        goto Exit;</p>

<p>    }</p>

<p>    pbyFileData =
(EC_T_BYTE*)OsMalloc(dwDataLen);</p>

<p>    pFile      
= OsFopen(szFileName, "rb");</p>

<p>    dwDataLen  
= OsFread(pbyFileData, 1, dwDataLen, pFile);</p>

<p>    fclose(pFile);</p>

<p> </p>

<p>    MbxTferDesc.dwMaxDataLen       = 0;</p>

<p>    MbxTferDesc.pbyMbxTferDescData = EC_NULL;</p>

<p>    pMbxTfer =
ecatMbxTferCreate(&amp;MbxTferDesc);</p>

<p>    if (EC_NULL != pMbxTfer)</p>

<p>    {</p>

<p>        pMbxTfer-&gt;dwClntId       = S_dwClntId;</p>

<p>        pMbxTfer-&gt;dwTferId       = 0xCCCCEEEE;</p>

<p>        pMbxTfer-&gt;dwDataLen      = dwDataLen;</p>

<p>        pMbxTfer-&gt;pbyMbxTferData =
pbyFileData;</p>

<p> </p>

<p>        dwRes = ecatFoeDownloadReq(pMbxTfer, 0,
szDownloadName, OsStrlen(szDownloadName), 0x00000000, EC_NOWAIT);</p>

<p>        if (EC_E_NOERROR != dwRes)</p>

<p>        {</p>

<p>            LogMsg("FoE download request
error (0x%08X)!", dwRes);</p>

<p>        }</p>

<p>        LogMsg("FoE download
started...");</p>

<p>        do</p>

<p>        {</p>

<p>            OsSleep(1);</p>

<p> </p>

<p>        } while(eMbxTferStatus_Idle !=
pMbxTfer-&gt;eTferStatus);</p>

<p> </p>

<p>        ecatMbxTferDelete(pMbxTfer);</p>

<p>    }</p>

<p>    OsFree(pbyFileData);</p>

<p>    if (dwRes != EC_E_NOERROR)</p>

<p>    {</p>

<p>        goto Exit;</p>

<p>    }</p>

<p>}</p>

handling for ecatFoeFileDownloadReq(): Already part of ecatNotification.cpp */</p>

<p>        LogError(

<p> </p>

<p>        pMbxTfer-&gt;eTferStatus =
eMbxTferStatus_Idle;</p>

<p>    }</p>

<p>    else if (pMbxTfer-&gt;eTferStatus !=
eMbxTferStatus_TferDone)</p>

%3d%%",

<p>    }</p>

eMbxTferStatus_Idle;</p>

<p>    } break;</p>
