<div>virtual EC_T_DWORD IoControl(EC_T_DWORD dwInstanceId, EC_T_DWORD dwCode, EC_T_IOCTLPARMS* pParms)</div>
<div>    {</div>
<div>        return emIoControl(dwInstanceId, dwCode, pParms);</div>
<div>    }</div>
</div><div>
<div>EC_T_DWORD CEmTestCommonApplication::Ioctl(</div>
<div>    EC_T_DWORD dwInstanceId,</div>
<div>    EC_T_DWORD dwCode,</div>
<div>    EC_T_VOID*      pbyInBuf,                       /*&lt; in  input data buffer */</div>
<div>    EC_T_DWORD      dwInBufSize,                    /*&lt; in  size of input data buffer in byte */</div>
<div>    EC_T_VOID*      pbyOutBuf,                      /*&lt; out output data buffer */</div>
<div>    EC_T_DWORD      dwOutBufSize,                   /*&lt; in  size of output data buffer in byte */</div>
<div>    EC_T_DWORD*     pdwNumOutData                   /*&lt; out number of output data bytes stored in output data buffer */</div>
<div>                                           )</div>
<div>{</div>
<div>    EC_T_IOCTLPARMS oIoCtlParms;</div>
<div>    EC_T_DWORD dwNumOutData = 0;</div>
<div><br></div>
<div>    oIoCtlParms.pbyInBuf = (EC_T_BYTE*)pbyInBuf;</div>
<div>    oIoCtlParms.dwInBufSize = dwInBufSize;</div>
<div>    oIoCtlParms.pbyOutBuf = (EC_T_BYTE*)pbyOutBuf;</div>
<div>    oIoCtlParms.dwOutBufSize = dwOutBufSize;</div>
<div>    oIoCtlParms.pdwNumOutData = (EC_NULL != pdwNumOutData) ? pdwNumOutData : &amp;dwNumOutData;</div>
<div>    return IoControl(dwInstanceId, dwCode, &amp;oIoCtlParms);</div>
<div>}</div>
</div><div>
<div>EC_T_DWORD CEmTestCommonApplication::SetFrameLossSimulation(EC_T_DWORD dwNumGoodFramesAfterStart, EC_T_DWORD dwFrameLossLikelihoodPpm,</div>
<div>                                                            EC_T_DWORD dwFixedLossNumGoodFrames, EC_T_DWORD dwFixedLossNumLostFrames,</div>
<div>                                                            EC_T_DWORD dwCode /* default: EC_IOCTL_SET_FRAME_LOSS_SIMULATION */)</div>
<div>{</div>
<div>    EC_T_DWORD adwFrameLossParams[4];</div>
<div>    m_bIgnoreFrameLoss = EC_TRUE;</div>
<div>#if (!defined EC_UNITTESTS)</div>
<div>    G_bTestBusDirty = EC_TRUE;</div>
<div>#endif</div>
<div><br></div>
<div>    adwFrameLossParams[0] = dwNumGoodFramesAfterStart;</div>
<div>    adwFrameLossParams[1] = dwFrameLossLikelihoodPpm;</div>
<div>    adwFrameLossParams[2] = dwFixedLossNumGoodFrames;</div>
<div>    adwFrameLossParams[3] = dwFixedLossNumLostFrames;</div>
<div><br></div>
<div>    return Ioctl(GetMasterId(), dwCode, (EC_T_BYTE*)&amp;adwFrameLossParams[0], 4 * sizeof(EC_T_DWORD), EC_NULL, EC_NULL, EC_NULL);</div>
<div>}</div>
